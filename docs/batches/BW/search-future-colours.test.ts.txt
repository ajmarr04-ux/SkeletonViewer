function run(order: string[], L: LayerColours[]) {
  const chosen: LayerColours[] = []; const res: any[] = []; let total = 0;
  for (const id of order) {
    const layers = [...L, ...chosen, { id, material: "#808080" }];
    const { checked } = pairsOf(layers, colourTable(layers, { dimmed: true }));
    const keys = vkeys(id);
    const mine = checked.filter(([a, b]) => a.layer === id || b.layer === id).map(([a, b]) => (a.layer === id ? [a, b] : [b, a]) as [Entry, Entry]);
    const self = mine.filter(([, b]) => b.layer === id).map(([a, b]) => [keys.indexOf(a.name), keys.indexOf(b.name)]);
    const others = mine.filter(([, b]) => b.layer !== id && !(process.env.NOREGION && b.role === "region")).map(([a, b]) => ({ a: keys.indexOf(a.name), bl: labs(b.colour), need: need(a.name, b.name) }));
    const sl = toLab(parseHex(seeds[id]!));
    const sorted = grid.map((x) => ({ x, d: deltaE2000(sl, x.v[0]![0]!) })).filter((y) => y.d < Number(process.env.R ?? 45)).sort((p, q) => p.d - q.d);
    let best: any;
    for (const { x, d } of sorted) {
      let ok = true, margin = 99;
      for (const o of others) { for (let k = 0; k < 3; k++) { const e = deltaE2000(x.v[o.a]![k]!, o.bl[k]!); if (e < o.need) { ok = false; break; } margin = Math.min(margin, e - o.need); } if (!ok) break; }
      if (ok) for (const [a, b] of self) { for (let k = 0; k < 3; k++) if (deltaE2000(x.v[a]![k]!, x.v[b]![k]!) < 8) { ok = false; break; } if (!ok) break; }
      if (ok) { best = { id, seed: seeds[id], c: x.c, moved: +d.toFixed(1), margin: +margin.toFixed(1) }; break; }
    }
    if (!best) { res.push({ id, none: true }); return { res, total: 1e9 }; }
    res.push(best); total += best.moved; chosen.push({ id, material: best.c });
  }
  return { res, total: +total.toFixed(1) };
}
function blockers(id: string, L: LayerColours[], radius: number) {
  const layers = [...L, { id, material: "#808080" }];
  const { checked } = pairsOf(layers, colourTable(layers, { dimmed: true }));
  const keys = vkeys(id);
  const mine = checked.filter(([a, b]) => a.layer === id || b.layer === id).map(([a, b]) => (a.layer === id ? [a, b] : [b, a]) as [Entry, Entry]).filter(([, b]) => b.layer !== id);
  const others = mine.map(([a, b]) => ({ n: a.name + "~" + b.name, a: keys.indexOf(a.name), bl: labs(b.colour), need: need(a.name, b.name) }));
  const sl = toLab(parseHex(seeds[id]!));
  const tally: Record<string, number> = {}; let n = 0; const only: Record<string, number> = {};
  for (const x of grid) { const d = deltaE2000(sl, x.v[0]![0]!); if (d > radius) continue; n++;
    const f = new Set<string>(); for (const o of others) for (let k = 0; k < 3; k++) if (deltaE2000(x.v[o.a]![k]!, o.bl[k]!) < o.need) f.add(process.env.BYVAR ? o.n.split("~")[0]!.replace(/@.*/, "") : o.n.split("~")[1]!);
    for (const b of f) tally[b] = (tally[b] ?? 0) + 1; if (f.size === 1) { const b = [...f][0]!; only[b] = (only[b] ?? 0) + 1; } }
  return { id, n, tally: Object.entries(tally).sort((a, b) => b[1] - a[1]).slice(0, 12), only };
}
